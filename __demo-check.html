const yamlComments = {
  highlight(code: string, lang: string) {
    if (lang !== "yaml") return null;
    return code
      .split(/(#[^\n]*)/)
      .filter((text) => text !== "")
      .map((text) => ({
        text,
        kind: text.startsWith("#") ? ("comment" as const) : ("plain" as const),
      }));
  },
};
</script>
